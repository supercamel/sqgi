static int generated_constant_division()
{
    int failures=0;
#if SQJIT_HAS_EXTERNAL_NATIVE
    ++native_test_groups;
    for(bool enabled : {false,true}) {
        SQVM *v=sq_open(64);
        SQJitContext &ctx=sqjit_context(v->_sharedstate);
        ctx.enabled=enabled;ctx.precise_exits=true;ctx.threshold=1;
        ctx.trace=ctx.trace_stats=false;sqjit_observe_enable(v);
        {
            SQObjectPtr a=SQArray::Create(v->_sharedstate,2);
            _array(a)->_values[0]=SQInteger(0);
            _array(a)->_values[1]=SQArray::Create(v->_sharedstate,1);
            const SQInteger divisors[]={1,2,7,64,97,997,1000003,2147483648LL,9223372036854775807LL};
            const SQInteger values[]={std::numeric_limits<SQInteger>::min(),-1001,-1,0,1,1001,
                std::numeric_limits<SQInteger>::max()};
            for(SQInteger divisor : divisors) for(bool remainder : {false,true}) {
                // An owning old destination exercises the checked helper path
                // without changing the constant proof or committed prefix.
                for(bool owning : {false,true}) {
                    std::string source="function constant_case(a,x) { a[0]=a[0]+1; ";
                    if(owning) source+="local q=a[1]; q=x";
                    else source+="local q=x";
                    source+=(remainder ? "%" : "/")+std::to_string((long long)divisor)+"; return q; }";
                    CHECK(run(v,source.c_str()),"compile signed constant arithmetic witness");
                    SQObjectPtr closure=named(v,_SC("constant_case"));
                    if(sq_type(closure)!=OT_CLOSURE) continue;
                    SQFunctionProto *proto=_closure(closure)->_function;
                    for(SQInteger n : values) {
                        const SQInteger count=_integer(_array(a)->_values[0]);
                        SQInteger got=0,want=remainder ? n%divisor : n/divisor;
                        CHECK(call(v,closure,a,n,got) && got==want &&
                            _integer(_array(a)->_values[0])==count+1,
                            "signed constant result and prefix match across extrema and aliases");
                    }
                    if(enabled) {
                        SQJitObservation route={};
                        CHECK(proto->_jit && proto->_jit->_entry && proto->_jit->_entry->_requires_vm_frame &&
                            sqjit_observe_proto(proto,route) && route.frame.successes>0,
                            "constant arithmetic executes generated canonical code");
                        if(proto->_jit && proto->_jit->_entry) {
                            const SQJitCode &code=proto->_jit->_entry->_code;
                            bool hardware=false,multiply_high=false;
                            for(SQInteger offset=0;offset<code.MappedSize();offset+=4) {
                                uint32_t word;memcpy(&word,(const unsigned char *)code.Entry()+offset,4);
                                hardware=hardware || (word&0xffe0fc00u)==0x9ac00c00u;
                                multiply_high=multiply_high || (word&0xffe0fc00u)==0x9b407c00u;
                            }
                            CHECK(!hardware,"proven positive divisor has no native hardware divide");
                            CHECK((divisor&(divisor-1))==0 || multiply_high,
                                "non-power constant executes emitted reciprocal multiplication");
                        }
                    }
                }
            }
            CHECK(run(v,_SC(
                "function joined_constant(a,x) { a[0]=a[0]+1; local d=7; if(x<0) d=11; return x/d; }\n"
                "function changing_constant(a,x) { local d=7,q=0; for(local i=0;i<3;i++) { a[0]=a[0]+1; q=x/d; d=11; } return q; }\n"
                "function zero_constant(a,x) { a[0]=a[0]+1; return x/0; }\n"
                "function negative_constant(a,x) { a[0]=a[0]+1; return x/-7; }\n"
                "function float_constant(a,x) { a[0]=a[0]+1; return x/7; }\n"
                "local a=[0]; if(float_constant(a,-17.5)!=-2.5 || a[0]!=1) throw \"constant float fallback\";\n"
                "local caught=false; try { zero_constant(a,11); } catch(e) { caught=true; }\n"
                "if(!caught || a[0]!=2) throw \"constant error prefix\";\n")),"constant float and zero-divisor fallbacks preserve semantics");
            if(enabled) {
                SQJitObservation floating={},zero={};
                CHECK(sqjit_observe_proto(_closure(named(v,_SC("float_constant")))->_function,floating) &&
                    floating.frame.successes>0,"constant divisor float fallback returns through native code");
                CHECK(sqjit_observe_proto(_closure(named(v,_SC("zero_constant")))->_function,zero) &&
                    zero.frame.side_exits==1,"constant zero error resumes once after the native committed prefix");
            }
            for(const SQChar *name : {_SC("joined_constant"),_SC("changing_constant"),_SC("negative_constant")}) {
                SQObjectPtr closure=named(v,name);
                for(SQInteger n : values) {
                    const bool changing=!scstrcmp(name,_SC("changing_constant"));
                    const bool negative=!scstrcmp(name,_SC("negative_constant"));
                    SQInteger divisor=negative ? -7 : changing || n<0 ? 11 : 7;
                    SQInteger got=0,count=_integer(_array(a)->_values[0]);
                    CHECK(call(v,closure,a,n,got) && got==n/divisor &&
                        _integer(_array(a)->_values[0])==count+(changing ? 3 : 1),
                        "joins and loop-carried divisors retain exact signed results and effects");
                }
                if(enabled) {
                    SQJitObservation route={};
                    CHECK(sqjit_observe_proto(_closure(closure)->_function,route) && route.frame.successes>0,
                        "dynamic and negative divisor controls actually execute native code");
                    SQFunctionProto *proto=_closure(closure)->_function;
                    if(proto->_jit && proto->_jit->_entry) {
                        const SQJitCode &code=proto->_jit->_entry->_code;
                        bool hardware=false;
                        for(SQInteger offset=0;offset<code.MappedSize();offset+=4) {
                            uint32_t word;memcpy(&word,(const unsigned char *)code.Entry()+offset,4);
                            hardware=hardware || (word&0xffe0fc00u)==0x9ac00c00u;
                        }
                        CHECK(hardware,"joins, loop-carried and negative divisors keep hardware fallback");
                    }
                }
            }
            CHECK(run(v,_SC("function private_loop_constant(n) { local marker=typeof n; local total=0; "
                "for(local i=0;i<n;i++) total+=(i-23)/7+(i-23)%11; return marker==\"integer\" ? total : 0; }")),
                "compile scalar loop with a supported body and unsupported whole-function prefix");
            for(SQInteger count : {0,1,73}) {
                SQInteger expected=0;
                for(SQInteger i=0;i<count;++i) expected+=(i-23)/7+(i-23)%11;
                std::string check="if(private_loop_constant("+std::to_string((long long)count)+")!="+
                    std::to_string((long long)expected)+") throw \"private constant loop\";";
                CHECK(run(v,check.c_str()),"private constant loop matches signed host arithmetic");
            }
            if(enabled) {
                SQFunctionProto *p=_closure(named(v,_SC("private_loop_constant")))->_function;
                SQJitObservation route={};
                CHECK(p->_jit && p->_jit->_loop_code.Entry() && sqjit_observe_proto(p,route) && route.loop.successes>0,
                    "constant arithmetic executes the separate private loop artifact");
                if(p->_jit && p->_jit->_loop_code.Entry()) {
                    const SQJitCode &code=p->_jit->_loop_code;
                    bool hardware=false,high=false;
                    for(SQInteger offset=0;offset<code.MappedSize();offset+=4) {
                        uint32_t word;memcpy(&word,(const unsigned char *)code.Entry()+offset,4);
                        hardware=hardware || (word&0xffe0fc00u)==0x9ac00c00u;
                        high=high || (word&0xffe0fc00u)==0x9b407c00u;
                    }
                    CHECK(!hardware && high,"private loop emits reciprocal arithmetic without hardware division");
                }
            }
            CHECK(!ctx.active_logs,"constant arithmetic leaves no active journal");
        }
        sq_close(v);
    }
    if(!failures) puts("[coverage] executed AArch64 constant signed division contracts");
#endif
    return failures;
}
