cmake_minimum_required(VERSION 3.16)
project(sqgi_jit_cross_check LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(SQGI_SOURCE /home/sam/Programming/sqgi)
file(GLOB core "${SQGI_SOURCE}/Squirrel3/squirrel/*.cpp"
    "${SQGI_SOURCE}/Squirrel3/sqstdlib/*.cpp"
    "${SQGI_SOURCE}/Squirrel3/squirrel/jit/*.cpp")
list(FILTER core EXCLUDE REGEX "sqjit_backend_(aarch64|none)")
if(SQGI_LOG_SOURCE)
    list(FILTER core EXCLUDE REGEX "/sqjit_write_log.cpp$")
    list(APPEND core "${SQGI_LOG_SOURCE}")
endif()
add_library(squirrel_check STATIC ${core})
target_include_directories(squirrel_check PUBLIC
    "${SQGI_SOURCE}/Squirrel3/include" "${SQGI_SOURCE}/Squirrel3/squirrel"
    "${SQGI_SOURCE}/Squirrel3/squirrel/jit")
target_compile_definitions(squirrel_check PUBLIC SQ_ENABLE_JIT _SQ64 SQUSEDOUBLE)
enable_testing()
foreach(test sqjit_write_log sqjit_member_plan sqjit_memory sqjit_leaf sqjit_object_plan sqjit_side_exit sqjit_contracts)
    add_executable(${test} "${SQGI_SOURCE}/test/cpp/test_${test}.cpp")
    target_link_libraries(${test} PRIVATE squirrel_check)
    add_test(NAME ${test} COMMAND qemu-x86_64 -L /usr/x86_64-linux-gnu $<TARGET_FILE:${test}>)
    set_tests_properties(${test} PROPERTIES WORKING_DIRECTORY "${SQGI_SOURCE}")
endforeach()
